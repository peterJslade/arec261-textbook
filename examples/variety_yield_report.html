Introduction

This report summarizes 2025 seeded acreage by variety, and average yield across all risk zones and years (2021-2025), for the major crops grown in Saskatchewan: canola, spring wheat, durum wheat, barley, oats, lentils, and peas.

For each crop, the acreage table below includes all varieties reported in 2025. The yield chart is restricted to “major” varieties, defined as those accounting for more than 2% of the crop’s total 2025 acreage.

# 2025 acreage by variety, sorted descending, with % of crop total
acreage_table_2025 <- function(crop_names) {
  variety_yields |>
    filter(Crop %in% crop_names, Year == 2025, !is.na(Acres)) |>
    group_by(Variety) |>
    summarise(Acres = sum(Acres), .groups = "drop") |>
    mutate(Pct_of_total = Acres / sum(Acres)) |>
    arrange(desc(Acres))
}

# Varieties accounting for more than 2% of 2025 acreage
major_varieties <- function(acreage_tbl) {
  acreage_tbl |>
    filter(Pct_of_total > 0.02) |>
    pull(Variety)
}

# Average yield (across all risk zones and years) for major varieties
yield_plot_major <- function(crop_names, varieties) {
  variety_yields |>
    filter(Crop %in% crop_names, Variety %in% varieties) |>
    group_by(Variety) |>
    summarise(mean_yield = mean(Yield, na.rm = TRUE), .groups = "drop") |>
    ggplot(aes(x = mean_yield, y = fct_reorder(Variety, mean_yield))) +
    geom_col() +
    labs(
      x = "Average yield (bu/acre)",
      y = "Variety",
      title = "Average yield of major varieties (2021-2025, all risk zones)"
    )
}
